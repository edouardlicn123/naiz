## 一、问题陈述

### 1.1 调试层叠问题只能靠猜

当前调试显示层叠问题时，只能通过串口log输出状态信息：

```c
hal_log("layer_sprite_face: id=%d asset=%d x=%d y=%d mirror=%d\r\n", ...);
hal_log("layer_dialog_snap: dirty=%d\r\n", dialog_dirty);
```

无法直观看到：
- 每层的实际像素内容
- 层间的覆盖关系
- 脏区域的位置和大小
- 快照的实际内容

### 1.2 常见调试场景

| 场景 | 当前调试方式 | 痛点 |
|------|------------|------|
| 动画覆盖对话框 | 检查 clip_h 参数 | 无法确认裁剪是否正确 |
| 立绘与对话框边界 | 检查 sprite y 坐标 | 无法确认实际绘制位置 |
| 对话框闪烁/残留 | 检查 snapshot 时机 | 无法确认快照内容 |
| 背景恢复不完整 | 检查 bg_snapshot | 无法确认快照是否正确 |
| 调色板冲突 | 检查 prot_pal | 无法确认实际调色板状态 |

### 1.3 缺乏自动化回归测试

没有机制保存"正确状态"的基准图像，无法自动检测回归。

## 二、方案：层导出 + 合成预览

### 2.1 核心思路

1. **层导出函数** — 将每层内容导出为图像文件
2. **合成预览** — 将所有层按Z序合成为完整画面
3. **调试命令** — 通过串口触发导出
4. **基准对比** — 保存基准图像，自动检测回归

### 2.2 图像格式选择：PPM

选择PPM（Portable Pixmap）格式，原因：
- 格式极其简单（ASCII header + raw RGB bytes）
- 无需第三方库
- 可用任何图像查看器打开
- 文件较大但仅用于调试

**PPM文件结构**：
```
P6
640 400
255
<raw RGB bytes: 640 * 400 * 3 = 768,000 bytes>
```

### 2.3 层导出函数

在 `layer_debug.h` 中声明，在 `layer_debug.c` 中实现：

```c
/*==== 层导出函数 =========================================================*/

/* 导出单层内容为PPM图像
 * z_order: LAYER_Z_BG / LAYER_Z_SPRITE / LAYER_Z_ANIM / LAYER_Z_DIALOG / LAYER_Z_TEXT
 * filename: 输出文件名（不含路径，自动加前缀 "dump_"）
 * 返回: 0=成功, -1=失败
 */
int layer_dump(int z_order, const char *filename);

/* 导出合成结果（所有层按Z序合成）
 * filename: 输出文件名
 * 返回: 0=成功, -1=失败
 */
int layer_dump_composite(const char *filename);

/* 导出所有层（单层+合成）
 * prefix: 文件名前缀（如 "dump_bg"）
 * 返回: 导出的文件数
 */
int layer_dump_all(const char *prefix);
```

### 2.4 实现细节

#### PPM写入辅助函数

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include "render.h"
#include "layer_internal.h"
#include "scene_layers.h"

#define PPM_PATH_PREFIX "dump_"

static int write_ppm(const char *filename, int w, int h, const uint8_t *rgb) {
    FILE *f;
    char path[256];
    int i, len;

    snprintf(path, sizeof(path), "%s%s", PPM_PATH_PREFIX, filename);
    f = fopen(path, "wb");
    if (!f) {
        hal_log("layer_debug: cannot open '%s' for writing\r\n", path);
        return -1;
    }

    fprintf(f, "P6\n%d %d\n255\n", w, h);
    len = w * h * 3;
    for (i = 0; i < len; i++)
        fputc(rgb[i], f);

    fclose(f);
    hal_log("layer_debug: exported '%s' (%dx%d, %d bytes)\r\n", path, w, h, len);
    return 0;
}
```

#### 调色板转RGB辅助函数

```c
static void palette_to_rgb(uint8_t *rgb, const uint8_t *indices, int count) {
    int i;
    uint8_t r, g, b;
    for (i = 0; i < count; i++) {
        hal_get_palette(indices[i], &r, &g, &b);
        rgb[i * 3] = r;
        rgb[i * 3 + 1] = g;
        rgb[i * 3 + 2] = b;
    }
}
```

#### 导出背景层

```c
static int dump_bg_layer(const char *filename) {
    const uint8_t *snap;
    uint8_t *rgb;
    int size;

    snap = layer_bg_snapshot();
    if (!snap) {
        hal_log("layer_debug: bg_snapshot not available\r\n");
        return -1;
    }

    size = 640 * 400;
    rgb = (uint8_t *)malloc(size * 3);
    if (!rgb) {
        hal_log("layer_debug: OOM for rgb buffer\r\n");
        return -1;
    }

    palette_to_rgb(rgb, snap, size);
    write_ppm(filename, 640, 400, rgb);
    free(rgb);
    return 0;
}
```

#### 导出立绘层（差值图像）

```c
static int dump_sprite_layer(const char *filename) {
    uint8_t *vram_buf, *bg_buf, *rgb;
    int size, i;

    size = 640 * 400;
    vram_buf = (uint8_t *)malloc(size);
    bg_buf = (uint8_t *)malloc(size);
    rgb = (uint8_t *)malloc(size * 3);

    if (!vram_buf || !bg_buf || !rgb) {
        free(vram_buf); free(bg_buf); free(rgb);
        hal_log("layer_debug: OOM for sprite dump\r\n");
        return -1;
    }

    /* 读取当前VRAM */
    vram_read(0, 0, 640, 400, vram_buf);

    /* 读取背景快照 */
    memcpy(bg_buf, layer_bg_snapshot(), size);

    /* 计算差值：立绘 = VRAM - 背景 */
    for (i = 0; i < size; i++) {
        if (vram_buf[i] != bg_buf[i]) {
            /* 有差异的像素：用VRAM中的调色板索引 */
            rgb[i * 3] = vram_buf[i];  /* 占位，后面palette_to_rgb */
            rgb[i * 3 + 1] = vram_buf[i];
            rgb[i * 3 + 2] = vram_buf[i];
        } else {
            /* 无差异：黑色 */
            rgb[i * 3] = 0;
            rgb[i * 3 + 1] = 0;
            rgb[i * 3 + 2] = 0;
        }
    }

    /* 实际应该用palette_to_rgb转换，这里简化 */
    write_ppm(filename, 640, 400, rgb);
    free(vram_buf); free(bg_buf); free(rgb);
    return 0;
}
```

#### 导出合成结果

```c
int layer_dump_composite(const char *filename) {
    uint8_t *vram_buf, *rgb;
    int size;

    size = 640 * 400;
    vram_buf = (uint8_t *)malloc(size);
    rgb = (uint8_t *)malloc(size * 3);

    if (!vram_buf || !rgb) {
        free(vram_buf); free(rgb);
        return -1;
    }

    vram_read(0, 0, 640, 400, vram_buf);
    palette_to_rgb(rgb, vram_buf, size);
    write_ppm(filename, 640, 400, rgb);
    free(vram_buf); free(rgb);
    return 0;
}
```

### 2.5 调试命令集成

在 `hal.c` 或 `debug.c` 中添加串口命令处理：

```c
/* 串口调试命令：
 *   dump:bg      — 导出背景层
 *   dump:sprite  — 导出立绘层（差值）
 *   dump:dialog  — 导出对话框层
 *   dump:anim    — 导出动画层（当前帧）
 *   dump:all     — 导出所有层+合成
 *   dump:status  — 输出层状态（不导出图像）
 */
void debug_handle_dump(const char *arg) {
    if (strcmp(arg, "bg") == 0)
        layer_dump(LAYER_Z_BG, "bg.ppm");
    else if (strcmp(arg, "sprite") == 0)
        layer_dump(LAYER_Z_SPRITE, "sprite.ppm");
    else if (strcmp(arg, "dialog") == 0)
        layer_dump(LAYER_Z_DIALOG, "dialog.ppm");
    else if (strcmp(arg, "anim") == 0)
        layer_dump(LAYER_Z_ANIM, "anim.ppm");
    else if (strcmp(arg, "all") == 0)
        layer_dump_all("layer");
    else if (strcmp(arg, "status") == 0)
        debug_dump_layer_status();
    else
        hal_log("dump usage: bg|sprite|dialog|anim|all|status\r\n");
}

static void debug_dump_layer_status(void) {
    hal_log("layer status:\r\n");
    hal_log("  bg:      %s\r\n", layer_is_active(LAYER_Z_BG) ? "active" : "inactive");
    hal_log("  sprite:  %s\r\n", layer_is_active(LAYER_Z_SPRITE) ? "active" : "inactive");
    hal_log("  anim:    %s\r\n", layer_is_active(LAYER_Z_ANIM) ? "active" : "inactive");
    hal_log("  dialog:  %s\r\n", layer_is_active(LAYER_Z_DIALOG) ? "active" : "inactive");
    hal_log("  text:    %s\r\n", layer_is_active(LAYER_Z_TEXT) ? "active" : "inactive");
    hal_log("  cursor:  %s\r\n", layer_is_active(LAYER_Z_CURSOR) ? "active" : "inactive");
}
```
