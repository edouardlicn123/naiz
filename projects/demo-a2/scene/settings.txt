; --- 对话框背景样式 ---
; dlgstyle: 对话框背景样式 (0-9)
;
;   取值范围  0 – 9，编码规则：
;     bit 0   = 抖动（0=实心填充, 1=75% PAT75 抖动）
;     bits 1-3 = 颜色索引（见下表）
;
;   颜色方案：
;     0 = 黑    (0x00, 0x00, 0x00)
;     1 = 蓝    (0x00, 0x00, 0xFF)
;     2 = 深红  (0x80, 0x00, 0x00)
;     3 = 绿    (0x00, 0x80, 0x00)
;     4 = 紫    (0x80, 0x00, 0x80)
;
;   常用值：
;     0 = 实心黑底         1 = 抖动黑底
;     2 = 实心蓝底         3 = 抖动蓝底
;     4 = 实心深红底       5 = 抖动深红底
;     6 = 实心绿底         7 = 抖动绿底
;     8 = 实心紫底         9 = 抖动紫底
;
;   建议：奇数位开启抖动，可减少纯色块的突兀感
dlgstyle=5

; --- 按钮配色方案 ---
; btnstyle: 按钮配色方案 (0-4)
;
;   0 = 黑    1 = 蓝    2 = 深红    3 = 绿    4 = 紫
;
;   影响按钮填充色、高亮色和阴影色（自动计算 75% 亮度）
btnstyle=2

; --- 语言 ---
; lang: 界面语言
;   由翻译系统 tr_init() 加载对应 .po 文件
;   引擎本身预设 eng/chi 两组字符串
;
;   可选值：eng | chi
;
;   eng — English 界面
;   chi — 中文界面
;
;   如需新增语言，需在 tools/po/ 下添加对应 .po 文件
lang=eng


; --- 黑花体（Blackletter）---
; blacktitle: 加载页标题是否使用黑花体 (0=关, 1=开)
;   仅当 lang 为 eng 等非 CJK 语言时生效；CJK 语言自动回退原字体
blacktitle=0

; blackdialog: 正文（对话/问题/存档等）ASCII 是否使用黑花体 16×16 (0=关, 1=开)
;   仅影响 ASCII 拉丁文字；CJK 走 CJK.DAT 不受影响；非 CJK 语言才生效
blackdialog=0
